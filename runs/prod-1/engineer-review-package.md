## Original description

> A dashboard for the software factory itself: see cycle time, cost, model mix, and defect/review-round trends from the ledger and findings data. The factory builds its own visibility layer.

## Platform

Local web app (browser), read-only over the factory's ledger/findings files. No backend server, no accounts; factory data stays local. Served via a local dev/preview server.

## Stack

React + Vite + TypeScript; Vitest + Testing Library (jsdom). Charts via a lightweight lib (e.g. recharts) or hand-rolled SVG. Reads JSONL ledger + findings files.

## Features

- headline cards
- per-story table
- trend views
- run/repo scope selector

## Acceptance criteria (per feature)

- **headline cards:** Show total tokens, total cost (or 'not recorded' when cost unknown), total escaped defects, mean review rounds, and process-overhead ratio across the selected scope. Each card renders a real computed number or 'not recorded' — never 0 for missing data. Cards update when scope changes.
- **per-story table:** One row per story in scope: story id, arm/run, coder/non-coder tokens, overhead ratio, wall time, human-min (or 'not recorded'), escaped defects, review rounds, model(s). Sortable by column. Nulls render 'not recorded', never 0. Matches factory/metrics.py harvest output.
- **trend views:** Charts of review-rounds-per-story and escaped-defects-per-story over the run sequence, plus model-mix breakdown (tokens per model). Charts render from real ledger data; empty scope shows a friendly empty state, not a blank axis. Tooltips show exact values.
- **run/repo scope selector:** Selector lists all runs/repos present in the ledger data; default = all. Changing scope recomputes cards/table/trends from that subset only. Selection persists across reload via 'factory-dash:prefs'. Selecting a scope with no data shows the empty state, not stale numbers.

## Data model

Reads existing schemas, defines no new write model. Ledger row (schema 0): {event, ts, story, role, repo, sha, model, tokens{in,out}, cost_usd, wall_s, human_min, outcome, detail, prev}. Finding row: {ts, repo, pr, round, source, klass, severity, summary, path, head_sha, escaped_gate, escaped_review, verified, disposition, fix_sha, gate_check}. Derived view models: StoryMetrics (coder/noncoder tokens, overhead_ratio, wall_s, human_min, defects, dispatches) and HeadlineSummary (totals across runs).

## Persistence

READ-ONLY: the dashboard never writes ledger/findings/state. It reads runs/*/ledger.jsonl and state/findings.jsonl from the local filesystem (or a bundled snapshot). No localStorage writes for factory data; UI prefs (selected run/repo) may persist in a separate localStorage key 'factory-dash:prefs'. On unreadable/corrupt file: show a clear error state for that source, never crash, never fabricate numbers (nulls render 'not recorded', per metrics.py convention).

## Look and feel

Match the factory's existing warm-paper theme (light + dark, contrast-audited AA). Consistent with the to-do demo palette; calm, readable, charts subordinate to numbers.

## Assumptions & resolved ambiguities

v1 is READ-ONLY visibility — no editing, no dispatch triggers, no board writes (those are the controller's job, OC-14). 'Cost' is tokens-only until a trusted price table exists (cost_usd often null/0 → render 'not recorded'). The dashboard reads a local snapshot of the ledger; live tailing/websockets are OUT of scope for v1. Per-dispatch drill-down and findings-mining views are deferred (owner chose the middle feature tier). No authentication (local-only).

const fmt = (n: number | null) => n === null ? "not recorded" : n.toLocaleString();

/** View 1 — headline cards. */
export function HeadlineCards() {
  const totalTokens = STORIES.reduce((a, s) => a + s.coder + s.non, 0);
  const escaped = STORIES.reduce((a, s) => a + s.defects, 0);
  const meanRounds = (STORIES.filter(s => s.arm === "factory").reduce((a, s) => a + s.rounds, 0) / 5).toFixed(1);
  const cards = [
    { label: "Total tokens", value: fmt(totalTokens) },
    { label: "Cost (USD)", value: "not recorded" },
    { label: "Escaped defects", value: String(escaped) },
    { label: "Mean review rounds", value: meanRounds },
    { label: "Overhead ratio", value: "0.94" },
  ];
  return (
    <div className="dash-card">
      <h1 className="dash-title">Factory dashboard</h1>
      <p className="dash-sub">All runs · all repos</p>
      <div className="dash-cards">
        {cards.map(c => (
          <div key={c.label} className="dash-stat">
            <span className="dash-stat-label">{c.label}</span>
            <span className="dash-stat-value">{c.value}</span>
          </div>
        ))}
      </div>
    </div>
  );
}

/** View 2 — per-story table (sortable). */
export function StoryTable() {
  const [sortKey, setSortKey] = useState<"rounds" | "defects" | "wall">("rounds");
  const rows = [...STORIES].sort((a, b) => (b[sortKey] as number) - (a[sortKey] as number));
  return (
    <div className="dash-card">
      <h1 className="dash-title">Per-story</h1>
      <table className="dash-table">
        <thead>
          <tr>
            <th>Story</th><th>Arm</th><th>Coder tok</th><th>Non-coder tok</th>
            <th><button className="dash-sort" onClick={() => setSortKey("rounds")}>Rounds</button></th>
            <th><button className="dash-sort" onClick={() => setSortKey("defects")}>Escaped</button></th>
            <th><button className="dash-sort" onClick={() => setSortKey("wall")}>Wall s</button></th>
            <th>Human min</th><th>Model</th>
          </tr>
        </thead>
        <tbody>
          {rows.map(s => (
            <tr key={s.id + s.arm}>
              <td>{s.id}</td><td>{s.arm}</td><td>{fmt(s.coder)}</td><td>{fmt(s.non)}</td>
              <td>{s.rounds}</td><td>{s.defects}</td><td>{s.wall}</td>
              <td>{fmt(s.human)}</td><td className="dash-model">{s.model}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

/** View 3 — trends (rounds + escaped defects + model mix). */
export function Trends() {
  const factory = STORIES.filter(s => s.arm === "factory");
  const maxR = Math.max(...factory.map(s => s.rounds));
  return (
    <div className="dash-card">
      <h1 className="dash-title">Trends</h1>
      <p className="dash-sub">Review rounds per story (factory arm)</p>
      <div className="dash-bars" role="img" aria-label="Review rounds per story">
        {factory.map(s => (

            <div className="dash-bar" style={{ height: `${(s.rounds / maxR) * 100}%` }} title={`${s.id}: ${s.rounds} rounds`} />
            <span className="dash-bar-label">{s.id.replace("TOD-", "")}</span>
          </div>
        ))}
      </div>
      <p className="dash-sub">Model mix (tokens)</p>
      <div className="dash-mix">
        <div className="dash-mix-row"><span>qwen4-max-preview</span><span className="dash-mix-bar" style={{ width: "70%" }} /><span>7.76M</span></div>
        <div className="dash-mix-row"><span>deepseek-v4-pro</span><span className="dash-mix-bar dash-mix-bar--alt" style={{ width: "30%" }} /><span>3.32M</span></div>
      </div>
    </div>
  );
}

/** View 4 — scope selector + empty state. */
export function ScopeSelector({ empty = false }: { empty?: boolean }) {
  const [scope, setScope] = useState("all");
  return (
    <div className="dash-card">
      <h1 className="dash-title">Scope</h1>
      <div className="dash-filters" role="group" aria-label="Select run scope">
        {["all", "todo-demo", "todo-control", "skel-5"].map(s => (
          <button key={s} className="dash-filter" aria-pressed={scope === s} onClick={() => setScope(s)}>{s}</button>
        ))}
      </div>
      {empty || scope === "skel-5" ? (
        <p className="dash-empty">No story data in this scope yet.</p>
      ) : (
        <p className="dash-sub">Showing {STORIES.length} stories across {scope === "all" ? "all runs" : scope}.</p>
      )}
    </div>
  );
}
