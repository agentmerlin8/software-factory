#!/usr/bin/env python3
"""DSH-1 thin data server: read-only JSONL access for the factory dashboard.

Python 3 stdlib only. Binds 127.0.0.1 only; opens source files for reading
only; never writes factory state. Endpoints (GET only, anything else 405):

  /api/runs            -> JSON list of run ids that have a ledger.jsonl
  /api/ledger?run=<id> -> JSON array of that run's ledger rows
  /api/findings        -> JSON array of findings rows

Sources are read fresh on every request from FACTORY_ROOT (default
~/Work/software-factory): runs/*/ledger.jsonl and state/findings.jsonl.
A corrupt/unreadable source yields 200 {"error": "<file>: <reason>"} —
never a crash, never partial or fabricated rows.
"""

import argparse
import json
import os
import re
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import parse_qs, urlparse

DEFAULT_FACTORY_ROOT = os.path.expanduser("~/Work/software-factory")
DEFAULT_PORT = 8765
# Run ids are plain directory names: no separators, no dot-only names.
RUN_ID_RE = re.compile(r"[A-Za-z0-9][A-Za-z0-9._-]*")


def read_jsonl(path):
    """Parse a JSONL file into a list of rows.

    Returns (rows, None) on success or (None, "<file>: <reason>") on any
    read/parse failure. All-or-nothing: a corrupt line means no rows.
    """
    try:
        with open(path, "r", encoding="utf-8") as fh:
            rows = []
            for line in fh:
                line = line.strip()
                if line:
                    rows.append(json.loads(line))
        return rows, None
    except (OSError, ValueError, UnicodeDecodeError) as exc:
        return None, f"{path}: {exc}"


class DashboardHandler(BaseHTTPRequestHandler):
    """GET-only handler; factory_root is injected by make_server."""

    factory_root: Path  # bound per-server by make_server()

    def _send_json(self, status, payload):
        body = json.dumps(payload).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _refuse_non_get(self):
        self._send_json(405, {"error": "method not allowed: read-only GET API"})

    do_POST = _refuse_non_get
    do_PUT = _refuse_non_get
    do_PATCH = _refuse_non_get
    do_DELETE = _refuse_non_get
    do_HEAD = _refuse_non_get
    do_OPTIONS = _refuse_non_get

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path == "/api/runs":
            self._runs()
        elif parsed.path == "/api/ledger":
            self._ledger(parse_qs(parsed.query))
        elif parsed.path == "/api/findings":
            self._findings()
        else:
            self._send_json(404, {"error": f"unknown endpoint: {parsed.path}"})

    def _runs(self):
        runs_dir = self.factory_root / "runs"
        try:
            ids = sorted(
                entry.name for entry in runs_dir.iterdir()
                if entry.is_dir() and (entry / "ledger.jsonl").is_file()
            )
        except OSError as exc:
            self._send_json(200, {"error": f"{runs_dir}: {exc}"})
            return
        self._send_json(200, ids)

    def _ledger(self, query):
        run_ids = query.get("run", [])
        if not run_ids or not run_ids[0]:
            self._send_json(400, {"error": "missing required query parameter: run"})
            return
        run = run_ids[0]
        # Refuse traversal: ids must be plain directory names (parse_qs has
        # already percent-decoded), and the resolved path must stay inside
        # runs/ as a second line of defense.
        if not RUN_ID_RE.fullmatch(run) or run in (".", ".."):
            self._send_json(400, {"error": f"invalid run id: {run!r}"})
            return
        runs_dir = (self.factory_root / "runs").resolve()
        ledger = (runs_dir / run / "ledger.jsonl").resolve()
        if not ledger.is_relative_to(runs_dir):
            self._send_json(400, {"error": f"invalid run id: {run!r}"})
            return
        if not ledger.is_file():
            self._send_json(404, {"error": f"unknown run: {run}"})
            return
        rows, error = read_jsonl(ledger)
        if error is not None:
            self._send_json(200, {"error": self._rel_error(error)})
            return
        self._send_json(200, rows)

    def _findings(self):
        findings = self.factory_root / "state" / "findings.jsonl"
        rows, error = read_jsonl(findings)
        if error is not None:
            self._send_json(200, {"error": self._rel_error(error)})
            return
        self._send_json(200, rows)

    def _rel_error(self, error):
        """Render '<file>: <reason>' with the file relative to FACTORY_ROOT."""
        prefix = str(self.factory_root) + os.sep
        if error.startswith(prefix):
            return error[len(prefix):]
        return error

    def log_message(self, format, *args):  # keep test/CI output quiet
        pass


def make_server(factory_root, port=DEFAULT_PORT):
    """Create a loopback-only server rooted at factory_root (a Path)."""
    handler = type("BoundHandler", (DashboardHandler,),
                   {"factory_root": Path(factory_root)})
    return ThreadingHTTPServer(("127.0.0.1", port), handler)


def main():
    parser = argparse.ArgumentParser(description="Factory dashboard read-only data server")
    parser.add_argument("--port", type=int, default=DEFAULT_PORT,
                        help=f"loopback port (default {DEFAULT_PORT})")
    args = parser.parse_args()
    root = Path(os.environ.get("FACTORY_ROOT", DEFAULT_FACTORY_ROOT))
    httpd = make_server(root, args.port)
    host, port = httpd.server_address[:2]
    print(f"factory dashboard data server: http://{host}:{port} (FACTORY_ROOT={root})")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        httpd.server_close()


if __name__ == "__main__":
    main()
"""DSH-1 server tests: happy paths, 405, 404, traversal, corrupt-file format.

Runs the real server (stdlib http.server) on an ephemeral loopback port
against a temporary FACTORY_ROOT fixture. Read-only: fixtures are created
by the tests, never by the server.
"""

import http.client
import json
import tempfile
import threading
import unittest
from pathlib import Path

import server

LEDGER_ROWS = [

def write_jsonl(path, rows):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text("".join(json.dumps(r) + "\n" for r in rows), encoding="utf-8")


class ServerTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls._tmp = tempfile.TemporaryDirectory()
        cls.root = Path(cls._tmp.name)
        write_jsonl(cls.root / "runs" / "run-a" / "ledger.jsonl", LEDGER_ROWS)
        (cls.root / "runs" / "run-no-ledger").mkdir(parents=True)  # excluded from /api/runs
        (cls.root / "runs" / "run-corrupt").mkdir(parents=True)
        (cls.root / "runs" / "run-corrupt" / "ledger.jsonl").write_text(
            '{"event": "ok"}\nnot json at all\n', encoding="utf-8")
        write_jsonl(cls.root / "state" / "findings.jsonl", FINDINGS_ROWS)
        cls.httpd = server.make_server(cls.root, port=0)
        cls.port = cls.httpd.server_address[1]
        cls.thread = threading.Thread(target=cls.httpd.serve_forever, daemon=True)
        cls.thread.start()

    @classmethod
    def tearDownClass(cls):
        cls.httpd.shutdown()
        cls.httpd.server_close()
        cls.thread.join(timeout=5)
        cls._tmp.cleanup()

    def request(self, method, path, body=None):
        conn = http.client.HTTPConnection("127.0.0.1", self.port, timeout=5)
        try:
            conn.request(method, path, body=body)
            resp = conn.getresponse()
            return resp.status, json.loads(resp.read().decode("utf-8"))
        finally:
            conn.close()

    def test_runs_list(self):
        status, data = self.request("GET", "/api/runs")
        self.assertEqual(status, 200)
        self.assertEqual(data, ["run-a", "run-corrupt"])  # only dirs with a ledger.jsonl

    def test_ledger_rows(self):
        status, data = self.request("GET", "/api/ledger?run=run-a")
        self.assertEqual(status, 200)
        self.assertEqual(data, LEDGER_ROWS)

    def test_findings_rows(self):
        status, data = self.request("GET", "/api/findings")
        self.assertEqual(status, 200)
        self.assertEqual(data, FINDINGS_ROWS)

    def test_non_get_405(self):
        for method, path in [("POST", "/api/runs"), ("POST", "/api/findings"),
                             ("PUT", "/api/ledger?run=run-a"), ("DELETE", "/api/runs")]:
            status, data = self.request(method, path, body="{}")
            self.assertEqual(status, 405, f"{method} {path}")
            self.assertIn("error", data)

    def test_unknown_run_404(self):
        status, data = self.request("GET", "/api/ledger?run=does-not-exist")
        self.assertEqual(status, 404)
        self.assertIn("error", data)
        self.assertIsInstance(data["error"], str)

    def test_traversal_refused(self):
        for run in ["../../state", "..%2F..%2Fstate", "..", "run-a/../../state"]:
            status, data = self.request("GET", f"/api/ledger?run={run}")
            self.assertIn(status, (400, 404), f"run={run!r} was not refused")
            self.assertIn("error", data)
            self.assertNotIsInstance(data, list)  # never leak rows

    def test_missing_run_param(self):
        status, data = self.request("GET", "/api/ledger")
        self.assertEqual(status, 400)
        self.assertIn("error", data)

    def test_corrupt_ledger_error_format(self):
        status, data = self.request("GET", "/api/ledger?run=run-corrupt")
        self.assertEqual(status, 200)
        self.assertIsInstance(data, dict)
        self.assertIn("error", data)
        self.assertTrue(data["error"].startswith("runs/run-corrupt/ledger.jsonl: "),
                        f"error must be '<file>: <reason>', got {data['error']!r}")
        self.assertNotIn("rows", data)  # no partial/fabricated rows

    def test_corrupt_findings_error_format(self):
        findings = self.root / "state" / "findings.jsonl"
        original = findings.read_text(encoding="utf-8")
        findings.write_text("{broken json\n", encoding="utf-8")
        try:
            status, data = self.request("GET", "/api/findings")
        finally:
            findings.write_text(original, encoding="utf-8")
        self.assertEqual(status, 200)
        self.assertIsInstance(data, dict)
        self.assertTrue(data["error"].startswith("state/findings.jsonl: "),
                        f"error must be '<file>: <reason>', got {data['error']!r}")

    def test_unreadable_source_error_format(self):
        findings = self.root / "state" / "findings.jsonl"
        findings.rename(findings.with_suffix(".jsonl.bak"))
        try:
            status, data = self.request("GET", "/api/findings")
        finally:
            findings.with_suffix(".jsonl.bak").rename(findings)
        self.assertEqual(status, 200)
        self.assertIsInstance(data, dict)
        self.assertTrue(data["error"].startswith("state/findings.jsonl: "))


if __name__ == "__main__":
    unittest.main()
