// harvestStories output (reuse, no reimplementation). Aggregation lives in
// the pure metrics helpers (AC 6); this component only renders them.
// Nulls -> 'not recorded' titles / 0-height bars, never a fabricated 0.
import { escapedBars, roundsBars, tokensByModel } from '../data/metrics';
import type { ModelMix, TrendBar } from '../data/metrics';
import type { RunLedger, StoryMetrics } from '../data/types';

/** Compact token totals for the mix rows (wireframe shows 7.76M / 3.32M).
 * Pure formatting; non-finite -> 'not recorded' (defense-in-depth). */
export function compactTokens(n: number): string {
  if (!Number.isFinite(n)) return 'not recorded';
  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`;
  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
  return String(n);
}

function BarChart({ bars, label, defect }: { bars: TrendBar[]; label: string; defect?: boolean }) {
  return (
    <div className="dash-bars" role="img" aria-label={label}>
      {bars.map((b) => (
        <div key={b.label} className="dash-bar-col">
          <div
            className={
              'dash-bar'
              + (defect ? ' dash-bar--defect' : '')
              + (b.value == null ? ' dash-bar--null' : '')
            }
            style={{ height: `${b.pct}%` }}
            title={b.title}
          />
          <span className="dash-bar-label">{b.label}</span>
        </div>
      ))}
    </div>
  );
}

function MixChart({ mix }: { mix: ModelMix[] }) {
  return (
    <div className="dash-mix" role="img" aria-label="Model mix (tokens)">
      {mix.map((m, i) => (
        <div key={m.model} className="dash-mix-row">
          <span>{m.model}</span>
          <span
            className={'dash-mix-bar' + (i % 2 === 1 ? ' dash-mix-bar--alt' : '')}
            style={{ width: `${m.pct}%` }}
            title={`${m.model}: ${m.tokens.toLocaleString()} tokens`}
          />
          <span>{compactTokens(m.tokens)}</span>
        </div>
      ))}
    </div>
  );
}

interface Props {
  stories: StoryMetrics[];
  /** Scoped run ledgers — the model-mix source (tokens grouped by row.model). */
  runs: RunLedger[];
}

export default function Trends({ stories, runs }: Props) {
  if (stories.length === 0) {
    return (
      <div className="dash-card">
        <h2 className="dash-title">Trends</h2>
        <p className="dash-empty">No story data in this scope yet.</p>
      </div>
    );
  }
  const rounds = roundsBars(stories);
  const escaped = escapedBars(stories);
  const mix = tokensByModel(runs);
  return (
    <div className="dash-card">
      <h2 className="dash-title">Trends</h2>
      <p className="dash-sub">Review rounds per story (factory arm)</p>
      {rounds.length === 0
        ? <p className="dash-empty">No factory-arm stories in this scope.</p>
        : <BarChart bars={rounds} label="Review rounds per story" />}
      <p className="dash-sub">Escaped defects per story (all arms)</p>
      <BarChart bars={escaped} label="Escaped defects per story" defect />
      <p className="dash-sub">Model mix (tokens)</p>
      {mix.length === 0
        ? <p className="dash-empty">No recorded model usage in this scope.</p>
        : <MixChart mix={mix} />}
      <p className="dash-note">Cycle-time trend deferred to v2 (wall time is in the table).</p>
    </div>
  );
}
// DSH-2 derived metrics — PURE functions over parsed rows (no fetch, no
// React). Formulas are single-sourced from the approved PRD and mirror
// factory/metrics.py harvest() semantics. Nulls stay null, never 0.
import type { Arm, FindingRow, LedgerTokens, RunLedger, Scope, StoryMetrics } from './types';

export const ALL = 'all';

// --- scope filtering (AC 5) --------------------------------------------------

export function selectRuns(runs: RunLedger[], run: string): RunLedger[] {
  return run === ALL ? runs : runs.filter((r) => r.run === run);
}

export function selectByRepo<T extends { repo?: string | null }>(rows: T[], repo: string): T[] {
  return repo === ALL ? rows : rows.filter((r) => r.repo === repo);
}

export function storiesInScope(stories: StoryMetrics[], scope: Scope): StoryMetrics[] {
  return stories.filter(
    (s) => (scope.run === ALL || s.run === scope.run) && (scope.repo === ALL || s.repo === scope.repo),
  );
}

// --- escaped defects (PRD formula 1) -----------------------------------------

/** (escaped_gate OR escaped_review) AND klass != 'false-positive' AND
 * disposition != 'rejected' AND summary does not start with 'RESOLVED'
 * (resolution markers are records, not defects). Source is findings.jsonl,
 * NOT ledger defect_found events (those count in-loop findings). */
export function isEscapedDefect(f: FindingRow): boolean {
  return Boolean(f.escaped_gate || f.escaped_review)
    && f.klass !== 'false-positive'
    && f.disposition !== 'rejected'
    && !(f.summary ?? '').startsWith('RESOLVED');
}

export function escapedDefectsByRepo(findings: FindingRow[]): Record<string, number> {
  // Null-prototype object: a repo named "constructor"/"toString"/etc. must
  // not collide with Object.prototype keys (Copilot DSH-2 round-3).
  const out: Record<string, number> = Object.create(null);
  for (const f of findings) {
    if (isEscapedDefect(f)) out[f.repo] = (out[f.repo] ?? 0) + 1;
  }
  return out;
}

// --- harvest (metrics.py semantics, PRD formula 2) ----------------------------

function tokenTotal(tokens: LedgerTokens | null | undefined): number {
  return (tokens?.in ?? 0) + (tokens?.out ?? 0);
}

// True only when an actual numeric usage value is present — a tokens object
// with null/undefined in AND out (or {}) records no usage (Copilot DSH-3
// round-5: missing usage must not display as recorded zero).
function hasUsage(tokens: LedgerTokens | null | undefined): boolean {
  return tokens != null
    && (typeof tokens.in === 'number' || typeof tokens.out === 'number');
}

function detailNumber(detail: Record<string, unknown> | null | undefined, key: string): number | null {
  const v = detail?.[key];
  return typeof v === 'number' && Number.isFinite(v) ? v : null;
}

interface Acc {
  story: string; run: string; arm: Arm; repo: string | null;
  coder: number; noncoder: number; wall: number | null; human: number | null;
  cost: number | null; dispatches: number; inLoopDefects: number;
  tokensRecorded: boolean;
  coderTokensRecorded: boolean; noncoderTokensRecorded: boolean;
  pr: number | null; prePrRounds: number | null; convergedRounds: number | null;
  models: string[];
}

/** Fold dispatch_completed (+ correction re-attribution by story/role),
 * human_merged, defect_found, pr_opened and loop_converged into per-story
 * metrics; rounds then come from findings per PRD formula 5. */
export function harvestStories(runs: RunLedger[], findings: FindingRow[] | null): StoryMetrics[] {
  const accs = new Map<string, Acc>();
  for (const runLedger of runs) {
    const run = runLedger.run;
    const rows = runLedger.rows;
    // Arm is a run-level fact: run_started.detail.arm === 'control' marks the
    // single-pass control arm; every other run is the factory arm.
    const arm: Arm = rows.some((r) => r.event === 'run_started' && r.detail?.arm === 'control')
      ? 'control' : 'factory';
    for (const row of rows) {
      const story = row.story;
      if (!story) continue;
      const key = run + '::' + story;
      let a = accs.get(key);
      if (!a) {
        a = {
          story, run, arm, repo: null, coder: 0, noncoder: 0, wall: null,
          human: null, cost: null, dispatches: 0, inLoopDefects: 0,
          tokensRecorded: false,
          coderTokensRecorded: false, noncoderTokensRecorded: false,
          pr: null, prePrRounds: null, convergedRounds: null, models: [],
        };
        accs.set(key, a);
      }
      if (row.repo && !a.repo) a.repo = row.repo;
      const ev = row.event;
      if (ev === 'dispatch_completed') {
        a.dispatches += 1;
        if (row.model && !a.models.includes(row.model)) a.models.push(row.model);
        const total = tokenTotal(row.tokens);
        if (hasUsage(row.tokens)) {
          a.tokensRecorded = true;
          if (row.role === 'coder') a.coderTokensRecorded = true;
          else a.noncoderTokensRecorded = true;
        }
        if (row.role === 'coder') a.coder += total; else a.noncoder += total;
        // Null-check, not truthy: a recorded ZERO duration/cost is a real
        // measurement and must accumulate (Copilot DSH-2 round-7).
        if (row.wall_s != null) a.wall = (a.wall ?? 0) + row.wall_s;
        if (row.cost_usd != null) a.cost = (a.cost ?? 0) + row.cost_usd;
        if (row.human_min != null) a.human = (a.human ?? 0) + row.human_min;
      } else if (ev === 'correction') {
        // Corrections re-attribute a dispatch's usage by story/role
        // (usage-path-bug recovery). Tokens live in detail.tokens.
        const d = row.detail ?? {};
        const corrTokens = d.tokens as LedgerTokens | null | undefined;
        // Mark recorded from non-null detail.tokens INDEPENDENTLY of the
        // accumulation guard — a zero-token correction is still recorded
        // usage (Copilot DSH-3 round-3).
        if (hasUsage(corrTokens)) {
          a.tokensRecorded = true;
          if (row.role === 'coder') a.coderTokensRecorded = true;
          else a.noncoderTokensRecorded = true;
        }
        const total = tokenTotal(corrTokens);
        if (total) {
          if (row.role === 'coder') a.coder += total; else a.noncoder += total;
          const c = detailNumber(d, 'cost_usd');
          if (c != null) a.cost = (a.cost ?? 0) + c;
        }
      } else if (ev === 'defect_found') {
        a.inLoopDefects += 1;
      } else if (ev === 'human_merged') {
        if (row.human_min != null) a.human = (a.human ?? 0) + row.human_min;
      } else if (ev === 'pr_opened') {
        const pr = detailNumber(row.detail, 'pr');
        if (pr != null) a.pr = pr;
        const pre = detailNumber(row.detail, 'pre_pr_rounds');
        if (pre != null) a.prePrRounds = pre;
      } else if (ev === 'loop_converged') {
        const pr = detailNumber(row.detail, 'pr');
        if (pr != null) a.pr = pr;
        const rounds = detailNumber(row.detail, 'rounds');
        if (rounds != null) a.convergedRounds = rounds;
      }
    }
  }
  const out: StoryMetrics[] = [];
  for (const a of accs.values()) {
    out.push({
      story: a.story, run: a.run, arm: a.arm, repo: a.repo,
      coderTokens: a.coder, noncoderTokens: a.noncoder,
      overheadRatio: a.coder === 0 ? null : a.noncoder / a.coder,
      wallS: a.wall, humanMin: a.human, costUsd: a.cost,
      dispatches: a.dispatches, inLoopDefects: a.inLoopDefects,
      tokensRecorded: a.tokensRecorded,
      coderTokensRecorded: a.coderTokensRecorded,
      noncoderTokensRecorded: a.noncoderTokensRecorded,
      rounds: storyRounds(a, findings ?? []),
      escapedDefects: storyEscapedDefects(a, findings),
      models: a.models,
    });
  }
  return out;
}

/** PRD formula (5): max round in findings for the story's PR + 1 pre-PR
 * round when present; else loop_converged detail.rounds; else null. */
function storyRounds(a: Acc, findings: FindingRow[]): number | null {
  if (a.pr != null && a.repo) {
    const rounds = findings
      .filter((f) => f.repo === a.repo && f.pr === a.pr)
      .map((f) => f.round)
      .filter((r) => typeof r === 'number' && Number.isFinite(r));
    if (rounds.length > 0) {
      return Math.max(...rounds) + (a.prePrRounds ?? 0);
    }
  }
  return a.convergedRounds;
}

/** DSH-4 (AC 5): per-story escaped defects — findings joined on repo+pr,
 * same join pattern as storyRounds, filtered by the shared isEscapedDefect
 * formula. Null when findings failed (source unavailable) or the story has
 * no recorded repo/pr to join on; a joined PR with zero escapes is a real 0. */
function storyEscapedDefects(a: Acc, findings: FindingRow[] | null): number | null {
  if (findings === null || a.pr == null || !a.repo) return null;
  return findings.filter(
    (f) => f.repo === a.repo && f.pr === a.pr && isEscapedDefect(f),
  ).length;
}

// --- headline aggregates (PRD formulas 3 and 4) -------------------------------

/** AGGREGATE sum(noncoder)/sum(coder) across scope — NOT the mean of
 * per-story ratios (matches metrics.py render_comparison headline).
 * Null when total coder tokens == 0. */
export function headlineOverhead(stories: StoryMetrics[]): number | null {
  let coder = 0;
  let noncoder = 0;
  for (const s of stories) {
    coder += s.coderTokens;
    noncoder += s.noncoderTokens;
  }
  return coder === 0 ? null : noncoder / coder;
}

// --- trends aggregation (DSH-5, pure — component renders from these) --------

/** Max-normalized percentage with a max=0/null guard: never NaN or
 * divide-by-zero. Null values render as 0-height (not-recorded) bars. */
export function barPct(value: number | null, max: number): number {
  if (value == null || max <= 0) return 0;
  return (value / max) * 100;
}

export interface TrendBar {
  label: string;
  value: number | null;
  pct: number;
  title: string;
}

/** AC 1: one bar per FACTORY-arm story, height max-normalized over recorded
 * rounds. Null rounds stay null -> 0-height + 'not recorded' title. */
export function roundsBars(stories: StoryMetrics[]): TrendBar[] {
  const factory = stories.filter((s) => s.arm === 'factory');
  const max = Math.max(0, ...factory.map((s) => s.rounds ?? 0));
  return factory.map((s) => ({
    label: s.story,
    value: s.rounds,
    pct: barPct(s.rounds, max),
    title: s.rounds == null ? `${s.story}: not recorded` : `${s.story}: ${s.rounds} rounds`,
  }));
}

/** AC 2: one bar per story (ALL arms), max-normalized with min divisor 1.
 * Control labels are distinguished; null escaped -> 0-height + title. */
export function escapedBars(stories: StoryMetrics[]): TrendBar[] {
  const max = Math.max(1, ...stories.map((s) => s.escapedDefects ?? 0));
  return stories.map((s) => {
    const escaped = s.escapedDefects ?? null;
    const label = s.arm === 'control' ? `ctl ${s.story}` : s.story;
    return {
      label,
      value: escaped,
      pct: barPct(escaped, max),
      title: escaped == null
        ? `${s.story} (${s.arm}): not recorded`
        : `${s.story} (${s.arm}): ${escaped} escaped`,
    };
  });
}

export interface ModelMix {
  model: string;
  tokens: number;
  pct: number;
}

/** AC 3: tokens aggregated by model across the scope (coder + non-coder),
 * from dispatch_completed + correction rows, ONLY rows with recorded usage
 * and a model name. Sorted desc by tokens; widths max-normalized (the top
 * model is 100%, so widths stay within [0,100]). A recorded zero is usage
 * and stays; a no-usage model is excluded, never shown as 0. */
export function tokensByModel(runs: RunLedger[]): ModelMix[] {
  const totals = new Map<string, number>();
  for (const runLedger of runs) {
    for (const row of runLedger.rows) {
      if (!row.model) continue;
      const tokens = row.event === 'dispatch_completed'
        ? row.tokens
        : row.event === 'correction'
          ? (row.detail?.tokens as LedgerTokens | null | undefined)
          : null;
      if (!hasUsage(tokens)) continue;
      totals.set(row.model, (totals.get(row.model) ?? 0) + tokenTotal(tokens));
    }
  }
  const entries = [...totals.entries()].sort((a, b) => b[1] - a[1]);
  const max = Math.max(0, ...entries.map(([, n]) => n));
  return entries.map(([model, tokens]) => ({ model, tokens, pct: barPct(tokens, max) }));
}

/** Mean over FACTORY-arm stories with recorded rounds (control excluded:
 * single-pass, not reviewable). Null when none recorded — never 0. */
export function meanReviewRounds(stories: StoryMetrics[]): number | null {
  const rounds = stories
    .filter((s) => s.arm === 'factory' && s.rounds != null)
    .map((s) => s.rounds as number);
  if (rounds.length === 0) return null;
  return rounds.reduce((sum, r) => sum + r, 0) / rounds.length;
}
/// <reference types="vite/client" />
// DSH-3 composite — scope selector ABOVE the data views; all views recompute
// from the selected run subset (AC 3). Prefs persist in 'factory-dash:prefs'
// (AC 4); empty scope and per-source errors get their own states (AC 5).
import { useEffect, useMemo, useState } from 'react';
import './index.css';
import HeadlineCards from './components/HeadlineCards';
import ScopeSelector from './components/ScopeSelector';
import StoryTable from './components/StoryTable';
import Trends from './components/Trends';
import { useDashboardData } from './hooks/useDashboardData';
import { ALL, escapedDefectsByRepo, harvestStories, selectRuns } from './data/metrics';

const PREFS_KEY = 'factory-dash:prefs';

function loadScope(): string {
  try {
    const prefs = JSON.parse(localStorage.getItem(PREFS_KEY) ?? '{}');
    return typeof prefs?.run === 'string' ? prefs.run : ALL;
  } catch {
    return ALL;
  }
}

export default function App() {
  const { loading, runs, ledgers, findings, errors } = useDashboardData();
  const [scope, setScope] = useState<string>(loadScope);

  const changeScope = (run: string) => {
    setScope(run);
    try {
      localStorage.setItem(PREFS_KEY, JSON.stringify({ run }));
    } catch { /* prefs are best-effort; the view still works */ }
  };

  // Reconcile a stale saved scope: if the persisted run no longer exists in
  // the discovered runs, fall back to ALL so data isn't hidden and a scope
  // button stays active (Copilot DSH-3 round-4).
  useEffect(() => {
    if (!loading && scope !== ALL && !runs.includes(scope)) {
      setScope(ALL);
    }
  }, [loading, runs, scope]);

  const stories = useMemo(
    // Pass findings through (null when the source failed): rounds fall back
    // to the ledger's loop_converged detail.rounds inside harvest — a valid
    // independent source (DSH-3 review F3) — while per-story escaped defects
    // stay null to signal findings unavailable (DSH-4 AC 5).
    () => harvestStories(selectRuns(ledgers, scope), findings),
    [ledgers, scope, findings],
  );
  // Escaped defects are per-repo (PRD formula 1); attribute them to the
  // repos present in the selected run scope. Unknown when findings failed.
  const escapedDefects = useMemo(() => {
    if (findings === null) return null;
    const byRepo = escapedDefectsByRepo(findings);
    if (scope === ALL) return Object.values(byRepo).reduce((sum, n) => sum + n, 0);
    const repos = new Set(stories.map((s) => s.repo).filter((r): r is string => r != null));
    let total = 0;
    for (const repo of repos) total += byRepo[repo] ?? 0;
    return total;
  }, [findings, scope, stories]);

  return (
    <div>
      <ScopeSelector runs={runs} value={scope} onChange={changeScope} />
      {errors.length > 0 && (
        <div className="dash-card">
          <div className="dash-error" role="alert">
            <strong>Could not read a data source.</strong>
            {errors.map((e) => <p key={e}>{e}</p>)}
          </div>
        </div>
      )}
      {loading ? (
        <div className="dash-card">
          <h1 className="dash-title">Factory dashboard</h1>
          <p className="dash-empty">Loading factory data…</p>
        </div>
      ) : stories.length === 0 ? (
        <div className="dash-card">
          <h1 className="dash-title">Factory dashboard</h1>
          <p className="dash-empty">No story data in this scope yet.</p>
        </div>
      ) : (
        <>
          <HeadlineCards
            stories={stories}
            escapedDefects={escapedDefects}
            sub={scope === ALL ? 'All runs' : scope}
          />
          <StoryTable stories={stories} />
          <Trends stories={stories} runs={selectRuns(ledgers, scope)} />
        </>
      )}
    </div>
  );
}
