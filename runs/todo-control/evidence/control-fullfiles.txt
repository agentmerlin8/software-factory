/**
 * store.ts — the ONLY module that touches localStorage (design §3).
 *
 * SKEL-1 scaffold: types + persistence helpers + subscription seam.
 * Story TOD-1 adds the first real operations; each story extends this
 * module. No story reads/writes localStorage directly.
 */

export interface Task {
  id: string;
  title: string;
  completed: boolean;
  createdAt: number;
}

export type Filter = "all" | "active" | "completed";

const STORAGE_KEY = "todo-demo:tasks";
const FILTER_KEY = "todo-demo:filter";

type Listener = () => void;
const listeners = new Set<Listener>();

export function subscribe(fn: Listener): () => void {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

function emit(): void {
  for (const fn of listeners) fn();
}

/** Validate a raw stored entry at the boundary (TOD-1 AC6 / TOD-2 AC5). */
function isTask(value: unknown): value is Task {
  if (typeof value !== "object" || value === null) return false;
  const t = value as Record<string, unknown>;
  return (
    typeof t.id === "string" &&
    typeof t.title === "string" &&
    typeof t.completed === "boolean" &&
    typeof t.createdAt === "number" &&
    Number.isFinite(t.createdAt)
  );
}

/** Tolerant read for RENDERING: any corruption yields an empty list. */
export function loadTasks(): Task[] {
  try {
    return readTasksStrict();
  } catch {
    return [];
  }
}

/**
 * Strict read for MUTATIONS (TOD-3 contract): storage ACCESS errors
 * propagate so callers can surface them; corrupt/malformed data still
 * degrades to [] (validated at the boundary).
 */
export function readTasksStrict(): Task[] {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (!raw) return [];
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    return [];
  }
  if (!Array.isArray(parsed)) return [];
  return parsed.filter(isTask);
}

export function saveTasks(tasks: Task[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
  emit();
}

/**
 * Append a task. Re-reads current storage at the write boundary and
 * merges by id (never overwrites from a stale snapshot), then returns
 * the COMMITTED snapshot for rendering — no post-write readback.
 * Throws on invalid title or storage failure; nothing is committed.
 */
export function addTask(title: string): Task[] {
  const trimmed = title.trim();
  if (!trimmed) throw new Error("Task title must not be empty");
  const current = readTasksStrict();
  const task: Task = {
    id: newId(),
    title: trimmed,
    completed: false,
    createdAt: Date.now(),
  };
  const committed = mergeById(current, [task]);
  saveTasks(committed);
  return committed;
}

/**
 * Flip `completed` for one task (TOD-3). Reads strictly, re-reads +
 * merges by id at the write boundary, returns the COMMITTED snapshot.
 * Absent id is a no-op (still returns the committed current snapshot).
 * Throws on storage access/write failure; nothing is committed then.
 */
export function toggleTask(id: string): Task[] {
  const current = readTasksStrict();
  const target = current.find((t) => t.id === id);
  if (!target) return current;
  const flipped: Task = { ...target, completed: !target.completed };
  const committed = mergeById(current, [flipped]);
  saveTasks(committed);
  return committed;
}

/**
 * Remove one task by id (TOD-4). Strict read, re-read at the write
 * boundary so concurrent commits survive, returns the COMMITTED
 * snapshot. Absent id is a no-op (no write). Throws on storage
 * access/write failure — the store keeps the task (no silent loss).
 */
export function deleteTask(id: string): Task[] {
  const current = readTasksStrict();
  if (!current.some((t) => t.id === id)) return current;
  const committed = current.filter((t) => t.id !== id);
  saveTasks(committed);
  return committed;
}

/** Merge by id: later entries win; order of first appearance is kept. */
function mergeById(base: Task[], incoming: Task[]): Task[] {
  const byId = new Map<string, Task>();
  for (const t of base) byId.set(t.id, t);
  for (const t of incoming) byId.set(t.id, t);
  return [...byId.values()];
}

/**
 * Pure newest-first ordering (TOD-2 AC1): createdAt descending.
 * Stable for equal timestamps — input order is preserved.
 */
export function sortNewestFirst(tasks: Task[]): Task[] {
  return [...tasks].sort((a, b) => b.createdAt - a.createdAt);
}

/**
 * Pure filter over tasks (TOD-5): never mutates input, never writes
 * task data. "all" returns a copy so callers can sort freely.
 */
export function filterTasks(tasks: Task[], filter: Filter): Task[] {
  if (filter === "active") return tasks.filter((t) => !t.completed);
  if (filter === "completed") return tasks.filter((t) => t.completed);
  return [...tasks];
}

export function loadFilter(): Filter {
  try {
    const raw = localStorage.getItem(FILTER_KEY);
    return raw === "active" || raw === "completed" ? raw : "all";
  } catch {
    // Carry-forward TOD-2 N3: a throwing getItem must not crash; fall back.
    return "all";
  }
}

export function saveFilter(filter: Filter): void {
  localStorage.setItem(FILTER_KEY, filter);
  emit();
}

export function newId(): string {
  return `${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;
}

import { useEffect, useRef, useState, type FormEvent } from "react";
import {
  addTask,
  deleteTask,
  filterTasks,
  loadFilter,
  loadTasks,
  saveFilter,
  sortNewestFirst,
  toggleTask,
  type Filter,
  type Task,
} from "./store";

/**
 * To-Do demo. All persistence goes through src/store.ts (design §3).
 * TOD-1: add. TOD-2: list. TOD-3: toggle. TOD-4: delete. TOD-5: filters.
 */
const FILTER_LABELS: Record<Filter, string> = {
  all: "All",
  active: "Active",
  completed: "Completed",
};

export default function App() {
  const [tasks, setTasks] = useState<Task[]>(() => loadTasks());
  const [filter, setFilter] = useState<Filter>(() => loadFilter());
  const [draft, setDraft] = useState("");
  const [rejected, setRejected] = useState(false);
  const [storageFailed, setStorageFailed] = useState(false);
  const [announcement, setAnnouncement] = useState("");
  const [pendingFocusId, setPendingFocusId] = useState<string | null>(null);
  // pendingFocusId: null = no request; "" = focus input; id = focus that row's delete button.
  const inputRef = useRef<HTMLInputElement>(null);
  const listRef = useRef<HTMLUListElement>(null);

  function clearAlerts() {
    setRejected(false);
    setStorageFailed(false);
  }

  function handleSubmit(e: FormEvent) {
    e.preventDefault();
    if (!draft.trim()) {
      setRejected(true);
      setStorageFailed(false);
      setAnnouncement("");
      inputRef.current?.focus();
      return;
    }
    try {
      const committed = addTask(draft);
      setTasks(committed);
      setDraft("");
      clearAlerts();
      const shown = filterTasks(committed, filter).length;
      setAnnouncement(
        `Added "${draft.trim()}". ${shown} tasks in ${FILTER_LABELS[filter]} view.`,
      );
      inputRef.current?.focus();
    } catch {
      setStorageFailed(true);
      setRejected(false);
      setAnnouncement("");
      inputRef.current?.focus();
    }
  }

  function handleToggle(task: Task, index: number) {
    try {
      const committed = toggleTask(task.id);
      setTasks(committed);
      clearAlerts();
      const done = committed.filter((t) => t.completed).length;
      const nowCompleted = committed.find((t) => t.id === task.id)?.completed;
      setAnnouncement(
        nowCompleted
          ? `Completed "${task.title}". ${done} of ${committed.length} tasks completed.`
          : `Marked "${task.title}" active again. ${done} of ${committed.length} tasks completed.`,
      );
      // If the row leaves the filtered view, move focus (TOD-4 pattern).
      const remaining = sortNewestFirst(filterTasks(committed, filter));
      if (!remaining.some((t) => t.id === task.id)) {
        const survivor = remaining[index] ?? remaining[index - 1];
        setPendingFocusId(survivor ? survivor.id : "");
      }
    } catch {
      // No silent unsaved state: tasks state is untouched (checkbox reverts).
      setStorageFailed(true);
      setRejected(false);
      setAnnouncement("");
    }
  }

  function handleDelete(task: Task, index: number) {
    try {
      const committed = deleteTask(task.id);
      const remaining = sortNewestFirst(filterTasks(committed, filter));
      // Focus a surviving control: next row, else previous, else input.
      const survivor = remaining[index] ?? remaining[index - 1];
      setPendingFocusId(survivor ? survivor.id : "");
      setTasks(committed);
      clearAlerts();
      setAnnouncement(
        `Deleted "${task.title}". ${remaining.length} tasks remaining in ${FILTER_LABELS[filter]} view.`,
      );
    } catch {
      // No silent data loss: state untouched, task stays visible.
      setStorageFailed(true);
      setRejected(false);
      setAnnouncement("");
    }
  }

  function handleFilter(next: Filter) {
    if (next === filter) return;
    try {
      // Writes ONLY the filter key — task data is never touched (AC4).
      saveFilter(next);
      setFilter(next);
      clearAlerts();
      const shown = filterTasks(tasks, next).length;
      setAnnouncement(`${FILTER_LABELS[next]} view: ${shown} tasks shown.`);
    } catch {
      // Keep the previous filter; no silent unsaved selection.
      setStorageFailed(true);
      setRejected(false);
      setAnnouncement("");
    }
  }

  useEffect(() => {
    if (pendingFocusId === null) return;
    if (pendingFocusId !== "") {
      const el = listRef.current?.querySelector<HTMLButtonElement>(
        `button[data-task-id="${CSS.escape(pendingFocusId)}"]`,
      );
      if (el) {
        el.focus();
        setPendingFocusId(null);
        return;
      }
    }
    // Lookup missed (row unmounted) or no survivor: input fallback.
    inputRef.current?.focus();
    setPendingFocusId(null);
  }, [pendingFocusId, tasks]);

  const visible = sortNewestFirst(filterTasks(tasks, filter));

  return (
    <main>
      <h1>To-Do</h1>
      <form onSubmit={handleSubmit} className="task-form">
        <label htmlFor="new-task">New task</label>
        <input
          id="new-task"
          ref={inputRef}
          value={draft}
          onChange={(e) => setDraft(e.target.value)}
          placeholder="What needs doing?"
          autoComplete="off"
        />
        <button type="submit">Add</button>
      </form>
      {(rejected || storageFailed) && (
        <p role="alert" className="error">
          {rejected
            ? "Please enter a task before adding."
            : "Could not save — storage is unavailable. Your draft was kept; try again."}
        </p>
      )}
      <p aria-live="polite" className="visually-hidden" data-testid="announcer">
        {announcement}
      </p>
      <div className="filters" role="group" aria-label="Filter tasks">
        {(Object.keys(FILTER_LABELS) as Filter[]).map((f) => (
          <button
            key={f}
            type="button"
            className="filter-button"
            aria-pressed={filter === f}
            onClick={() => handleFilter(f)}
          >
            {FILTER_LABELS[f]}
          </button>
        ))}
      </div>
      <p data-testid="task-count">{visible.length} tasks</p>
      {visible.length === 0 ? (
        tasks.length === 0 ? (
          <p>Nothing to do yet — add your first task above.</p>
        ) : (
          <p>No {FILTER_LABELS[filter].toLowerCase()} tasks right now.</p>
        )
      ) : (
        <ul className="task-list" ref={listRef}>
          {visible.map((task, index) => (
            <li key={task.id}>
              <input
                type="checkbox"
                className="task-toggle"
                checked={task.completed}
                onChange={() => handleToggle(task, index)}
                aria-label={`Mark "${task.title}" as ${task.completed ? "active" : "completed"}`}
              />
              <span className={task.completed ? "task-title completed" : "task-title"}>
                {task.title}
              </span>
              <button
                type="button"
                className="task-delete"
                data-task-id={task.id}
                onClick={() => handleDelete(task, index)}
                aria-label={`Delete "${task.title}"`}
              >
                Delete
              </button>
            </li>
          ))}
        </ul>
      )}
    </main>
  );
}
