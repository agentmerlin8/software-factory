/**
 * store.ts — the ONLY module that touches localStorage (design §3).
 *
 * SKEL-1 scaffold: types + persistence helpers + subscription seam.
 * Story TOD-1 adds the first real operations; each story extends this
 * module. No story reads/writes localStorage directly.
 */

export interface Task {
  id: string;
  title: string;
  completed: boolean;
  createdAt: number;
}

export type Filter = "all" | "active" | "completed";

const STORAGE_KEY = "todo-demo:tasks";
const FILTER_KEY = "todo-demo:filter";

type Listener = () => void;
const listeners = new Set<Listener>();

export function subscribe(fn: Listener): () => void {
  listeners.add(fn);
  return () => listeners.delete(fn);
}

function emit(): void {
  for (const fn of listeners) fn();
}

/**
 * Runtime guard for one stored entry. Storage is an UNTRUSTED boundary: the
 * JSON may be written by an older version, another tab, or a user with
 * devtools. `Array.isArray` alone lets `[null]` or `[{}]` reach the render
 * and throw during mount, so every field is validated here, once.
 */
export function isTask(value: unknown): value is Task {
  if (typeof value !== "object" || value === null) return false;
  const t = value as Record<string, unknown>;
  return (
    typeof t.id === "string" &&
    t.id.length > 0 &&
    typeof t.title === "string" &&
    typeof t.completed === "boolean" &&
    typeof t.createdAt === "number" &&
    Number.isFinite(t.createdAt)
  );
}

export function loadTasks(): Task[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const parsed: unknown = JSON.parse(raw);
    if (!Array.isArray(parsed)) return [];
    // Drop malformed entries instead of crashing the mount. Dedupe by id so a
    // corrupt store cannot produce duplicate React keys.
    const seen = new Set<string>();
    const tasks: Task[] = [];
    for (const entry of parsed) {
      if (!isTask(entry) || seen.has(entry.id)) continue;
      seen.add(entry.id);
      tasks.push(entry);
    }
    return tasks;
  } catch {
    return [];
  }
}

export function saveTasks(tasks: Task[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
  emit();
}

/**
 * TOD-2 — newest-first ordering. Pure: returns a new array, never mutates
 * the input (loadTasks hands out fresh arrays, but callers may hold their
 * own). Array.prototype.sort is stable (ES2019+), so tasks sharing a
 * createdAt keep their stored order — no secondary tiebreaker needed.
 */
export function sortNewestFirst(tasks: Task[]): Task[] {
  return [...tasks].sort((a, b) => b.createdAt - a.createdAt);
}

/**
 * Append records with the read and the write in ONE synchronous step.
 *
 * A caller that reads, does work, then writes always has a window where
 * another tab can commit; merging BEFORE the write does not close it (the
 * window is between the merge-read and setItem). Here the current state is
 * read immediately before setItem with nothing in between, so a concurrent
 * commit is either already visible (and preserved) or lands after ours.
 *
 * localStorage has no compare-and-swap, so this narrows the window to the
 * minimum the platform allows rather than eliminating it. Cross-tab
 * atomicity would need the Web Locks API; that is a deliberate follow-up,
 * not something this function claims to provide.
 */
export function appendTasks(incoming: Task[]): Task[] {
  const current = loadTasks();
  const byId = new Map(current.map((t) => [t.id, t]));
  for (const t of incoming) {
    if (!byId.has(t.id)) byId.set(t.id, t);
  }
  const merged = [...byId.values()];
  localStorage.setItem(STORAGE_KEY, JSON.stringify(merged));
  emit();
  return merged;
}

export function loadFilter(): Filter {
  const raw = localStorage.getItem(FILTER_KEY);
  return raw === "active" || raw === "completed" ? raw : "all";
}

export function saveFilter(filter: Filter): void {
  localStorage.setItem(FILTER_KEY, filter);
  emit();
}

export function newId(): string {
  return `${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`;
}

/**
 * TOD-1 — add a task. Trims the title; empty/whitespace-only titles are
 * rejected (returns null, store untouched, no notification). Persistence
 * and notification go through the same seam every later story extends.
 * Returns the created task so callers can update their view immediately.
 */
/**
 * Append a task without clobbering a concurrent writer.
 *
 * localStorage has no compare-and-swap, so a plain read-modify-write loses
 * updates when two tabs interleave (both read the same list, both write, the
 * second erases the first). We cannot make the pair atomic, but we can make
 * the WRITE a merge: re-read immediately before persisting and union by id.
 * That narrows the window to a single synchronous read+write and, crucially,
 * never drops a record another tab already committed.
 */
export function addTask(title: string): Task | null {
  const trimmed = title.trim();
  if (!trimmed) return null;
  const task: Task = {
    id: newId(),
    title: trimmed,
    completed: false,
    createdAt: Date.now(),
  };
  // Single synchronous read+merge+write; see appendTasks for the guarantee
  // and its limit.
  appendTasks([task]);
  return task;
}

/**
 * TOD-3 — flip one task's `completed` flag. Reads the current store
 * immediately before the write (same minimum-window discipline as
 * appendTasks), so a concurrent commit from another tab is preserved.
 * Returns the updated task, or null when the id is unknown (no write, no
 * notification). A storage failure propagates BEFORE emit and before any
 * caller can render the new state — nothing unsaved is ever announced
 * (TOD-3 AC5).
 */
export function toggleTask(id: string): Task | null {
  const current = loadTasks();
  const index = current.findIndex((t) => t.id === id);
  if (index === -1) return null;
  const toggled: Task = {
    ...current[index],
    completed: !current[index].completed,
  };
  const next = [...current];
  next[index] = toggled;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
  emit();
  return toggled;
}

describe("concurrent writers (round-8 regression)", () => {
  /**
   * Lost update = another tab commits between our READ and our WRITE.
   *
   * Two earlier attempts at this test were wrong, and both are worth
   * remembering:
   *  1. Injecting into our own outgoing setItem payload — the other task
   *     survives even with buggy code, so the test proves nothing.
   *  2. Injecting on getItem but RETURNING the pre-injection value — the
   *     implementation then correctly reads an empty store, so the test
   *     fails against correct code.
   *
   * Correct shape: seed the other tab's commit so it is VISIBLE to our read,
   * then assert our write preserves it. A stale-snapshot implementation
   * (read once at module/arg time, write later) drops it.
   */
  it("preserves a task another tab committed before our write", () => {
    // Another tab has already committed its task.
    saveTasks([
      { id: "other-tab", title: "FromOtherTab", completed: false, createdAt: 1 },
    ]);
    // Our add must merge with what is actually persisted, not clobber it.
    addTask("Mine");
    const titles = loadTasks().map((t) => t.title);
    expect(titles).toContain("FromOtherTab");
    expect(titles).toContain("Mine");
  });

  it("merges rather than overwrites when the store changes mid-operation", () => {
    // Simulate a commit landing after the caller took its snapshot: the
    // caller hands appendTasks only ITS record, and the store already holds
    // another tab's. A plain saveTasks([...snapshot, mine]) loses the other.
    saveTasks([
      { id: "a", title: "TabA", completed: false, createdAt: 1 },
    ]);
    const mine = { id: "b", title: "TabB", completed: false, createdAt: 2 };
    appendTasks([mine]);
    expect(loadTasks().map((t) => t.title).sort()).toEqual(["TabA", "TabB"]);
  });

  it("appendTasks is idempotent on a repeated id", () => {
    const t = { id: "x", title: "Once", completed: false, createdAt: 1 };
    appendTasks([t]);
    appendTasks([t]);
    expect(loadTasks()).toHaveLength(1);
  });

  it("re-reads at the write boundary rather than trusting a stale snapshot", () => {
    // Another tab persists BEFORE our add begins.
    saveTasks([{ id: "pre", title: "PreExisting", completed: false, createdAt: 1 }]);
    addTask("New");
    expect(loadTasks().map((t) => t.title)).toEqual(["PreExisting", "New"]);
  });

  it("never duplicates a task id under repeated adds", () => {
    addTask("A");
    addTask("B");
    addTask("C");
    const ids = loadTasks().map((t) => t.id);
    expect(new Set(ids).size).toBe(ids.length);
  });

import { useRef, useState } from "react";
import type { FormEvent } from "react";
import { addTask, loadTasks, sortNewestFirst, toggleTask, type Task } from "./store";

/**
 * TOD-1 — add a task; TOD-2 — list tasks; TOD-3 — complete/uncomplete.
 * Form + list on the store seam (src/store.ts is the only module touching
 * localStorage). Ordering and the empty state live here; deleting and
 * filtering land in TOD-4…5.
 */
export default function App() {
  const [tasks, setTasks] = useState<Task[]>(() => sortNewestFirst(loadTasks()));
  const [title, setTitle] = useState("");
  const [rejected, setRejected] = useState(false);
  const [storageFailed, setStorageFailed] = useState(false);
  // Polite live region: every successful mutation is announced, not just
  // errors. A sighted user sees the list change; a screen-reader user must
  // be told, or the action is silent (round-5 finding).
  const [announcement, setAnnouncement] = useState("");
  const inputRef = useRef<HTMLInputElement>(null);

  function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    try {
      const task = addTask(title);
      if (!task) {
        // AC2: empty/whitespace rejected — no task created, input keeps focus.
        setRejected(true);
        inputRef.current?.focus();
        return;
      }
      // Read back from the store, not from React state: another tab may have
      // persisted tasks since this component last rendered. Spreading stale
      // `tasks` would silently drop them until reload. Order at the view
      // boundary so adds and reloads both render newest-first (TOD-2 AC1).
      const next = sortNewestFirst(loadTasks());
      setTasks(next);
      setTitle("");
      setRejected(false);
      setStorageFailed(false);
      // Include the running total: identical live-region text does NOT
      // re-trigger an announcement, so two tasks with the same title would
      // otherwise announce only once. The count is both distinct per add and
      // useful information ("3 tasks" tells the user where they stand).
      setAnnouncement(
        `Added ${task.title}. ${next.length} ${next.length === 1 ? "task" : "tasks"}.`,
      );
      inputRef.current?.focus();
    } catch {
      setStorageFailed(true);
      inputRef.current?.focus();
    }
  }

  function handleToggle(id: string) {
    try {
      const toggled = toggleTask(id);
      if (!toggled) return;
      // Read back from the store (same discipline as add): another tab may
      // have committed tasks since this component last rendered.
      setTasks(sortNewestFirst(loadTasks()));
      setStorageFailed(false);
      // Distinct text per direction, so consecutive toggles always differ
      // and the live region re-announces (identical text does not).
      setAnnouncement(
        toggled.completed
          ? `Completed ${toggled.title}.`
          : `Marked ${toggled.title} as active.`,
      );
    } catch {
      // AC5: `tasks` is untouched, so the controlled checkbox reverts to the
      // persisted state on re-render — the UI never shows an unsaved flip.
      // The failure surfaces on the same assertive alert the add form uses.
      setStorageFailed(true);
    }
  }

  return (
    <main className="app">
      <header className="app-header">
        <p className="eyebrow">local-first · no account</p>
        <h1>To-Do</h1>
        <p className="tagline">Small steps, written down.</p>
      </header>

      <form className="add-form" onSubmit={handleSubmit} noValidate>
        <label className="visually-hidden" htmlFor="new-task">
          New task
        </label>
        <div className={rejected ? "add-row add-row--rejected" : "add-row"}>
          <input
            id="new-task"
            ref={inputRef}
            className="add-input"
            type="text"
            value={title}
            onChange={(event) => {
              setTitle(event.target.value);
              if (rejected) setRejected(false);
              if (storageFailed) setStorageFailed(false);
            }}
            placeholder="What needs doing?"
            autoComplete="off"
            autoFocus
            aria-invalid={rejected}
            aria-describedby="add-hint"
          />
          <button className="add-button" type="submit">
            Add
          </button>
        </div>
        <p
          id="add-hint"
          className={
            rejected || storageFailed
              ? "add-hint add-hint--error"
              : "add-hint"
          }
          role={rejected || storageFailed ? "alert" : undefined}
        >
          {rejected
            ? "Give the task a name — blank tasks don’t count."
            : storageFailed
              ? "Task couldn’t be saved. Check browser storage and try again."
              : "Press Enter or hit Add. Everything stays in this browser."}
        </p>
      </form>

      {/* Always mounted: a live region added at the same time as its text is
          frequently missed by screen readers. */}
      <p className="visually-hidden" role="status" aria-live="polite" aria-atomic="true">
        {announcement}
      </p>

      <section className="list-section" aria-label="Tasks">
        <p data-testid="task-count" className="task-count">
          {tasks.length === 1 ? "1 task" : `${tasks.length} tasks`}
        </p>
        {tasks.length > 0 ? (
          <ul className="task-list">
            {tasks.map((task) => (
              <li
                key={task.id}
                className={
                  task.completed ? "task-item task-item--completed" : "task-item"
                }
              >
                {/* TOD-3 AC1: the checkbox takes the marker slot the decorative
                    dot held. Controlled by the store-backed state, so a failed

          // never a blank section. Reuses the theme's .tagline — no new CSS.
          <p data-testid="empty-state" className="tagline">

          </p>
        )}
      </section>

        <p>Saved to this browser only — reload-safe, account-free.</p>
